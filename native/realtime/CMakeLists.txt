cmake_minimum_required(VERSION 3.18)
project(exe_realtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(AGBPLAY_SOURCE "" CACHE PATH "Pinned agbplay checkout")
set(LOCAL_DEPS "" CACHE PATH "Optional locally extracted dependency prefix")
if(LOCAL_DEPS)
  list(PREPEND CMAKE_PREFIX_PATH "${LOCAL_DEPS}/usr")
endif()
find_path(FMT_INCLUDE fmt/format.h HINTS "${LOCAL_DEPS}/usr/include" REQUIRED)
find_path(BOOST_INCLUDE boost/math/special_functions/sinc.hpp HINTS "${LOCAL_DEPS}/usr/include" REQUIRED)
find_path(ZIP_INCLUDE zip.h HINTS "${LOCAL_DEPS}/usr/include" REQUIRED)
find_library(FMT_LIBRARY fmt HINTS "${LOCAL_DEPS}/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}" REQUIRED)
find_library(ZIP_LIBRARY zip HINTS "${LOCAL_DEPS}/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}" REQUIRED)
find_package(ZLIB REQUIRED)
set(CORE "${AGBPLAY_SOURCE}/src/agbplay")
set(CORE_NAMES CGBPatterns Debug FileReader Gsf LoudnessCalculator MP2KChn MP2KChnPCM MP2KChnPSG MP2KContext MP2KPlayer MP2KScanner MP2KTrack Resampler ResamplerAVX2 ReverbEffect Rom SequenceReader SoundMixer Types Xcept)
foreach(NAME IN LISTS CORE_NAMES)
  list(APPEND CORE_SOURCES "${CORE}/${NAME}.cpp")
endforeach()
add_library(exe_agbplay SHARED ${CORE_SOURCES})
target_include_directories(exe_agbplay PUBLIC "${CORE}" "${FMT_INCLUDE}" PRIVATE "${BOOST_INCLUDE}" "${ZIP_INCLUDE}")
target_link_libraries(exe_agbplay PRIVATE "${FMT_LIBRARY}" "${ZIP_LIBRARY}" ZLIB::ZLIB)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|i.86")
  set_source_files_properties("${CORE}/ResamplerAVX2.cpp" PROPERTIES COMPILE_OPTIONS "-mavx2")
endif()
add_executable(exe-realtime main.cpp)
target_link_libraries(exe-realtime PRIVATE exe_agbplay "${FMT_LIBRARY}")
set_target_properties(exe-realtime PROPERTIES BUILD_RPATH "$ORIGIN")
